# COMANDOS PARA CREAR CARPETAS EN EL VPS (ejecutar en SSH)

# OPCIÓN 1: CREAR CARPETAS DESDE FUERA DEL CONTENEDOR (EN EL HOST)
# Esto crea las carpetas en el volumen persistente de Docker

# Crear carpetas songs y events con permisos completos:
docker exec cgplayer-app mkdir -p /app/backend/uploads/songs
docker exec cgplayer-app mkdir -p /app/backend/uploads/events
docker exec cgplayer-app chmod 777 /app/backend/uploads/songs
docker exec cgplayer-app chmod 777 /app/backend/uploads/events

# Verificar que las carpetas existen y tienen los permisos correctos:
docker exec cgplayer-app ls -la /app/backend/uploads/

# OPCIÓN 2: CREAR CARPETAS DESDE DENTRO DEL CONTENEDOR
# Entrar al contenedor:
docker exec -it cgplayer-app /bin/sh

# Una vez dentro del contenedor:
cd /app/backend/uploads
mkdir -p songs events
chmod 777 songs events
ls -la

# Salir del contenedor:
exit

# OPCIÓN 3: COMANDO TODO EN UNO (MÁS RÁPIDO)
docker exec cgplayer-app /bin/sh -c 'mkdir -p /app/backend/uploads/songs /app/backend/uploads/events && chmod 777 /app/backend/uploads/songs /app/backend/uploads/events && ls -la /app/backend/uploads/ && echo "✅ Carpetas creadas con permisos 777"'

# VERIFICACIÓN ADICIONAL:
# Ver estructura completa de uploads:
docker exec cgplayer-app find /app/backend/uploads -type d -exec ls -ld {} \;

# Ver permisos específicos:
docker exec cgplayer-app stat /app/backend/uploads/songs
docker exec cgplayer-app stat /app/backend/uploads/events
