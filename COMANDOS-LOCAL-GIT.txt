# COMANDOS PARA EJECUTAR EN TU LOCAL (uno por uno)

# 1. Ver qué archivos de uploads están siendo trackeados actualmente:
git ls-files backend/uploads/

# 2. Remover del tracking todos los archivos de uploads (MANTENIÉNDOLOS LOCALMENTE):
git rm -r --cached backend/uploads/songs/* 2>/dev/null || true
git rm -r --cached backend/uploads/events/* 2>/dev/null || true

# 3. Remover imágenes específicas (conservar .gitkeep y README.md):
git ls-files backend/uploads/images/playlists/ | grep -v '.gitkeep' | grep -v 'README.md' | xargs -r git rm --cached
git ls-files backend/uploads/images/profiles/ | grep -v '.gitkeep' | xargs -r git rm --cached

# 4. Verificar que los archivos ya no están siendo trackeados:
git ls-files backend/uploads/

# 5. Verificar que los archivos siguen existiendo localmente:
ls -la backend/uploads/songs/
ls -la backend/uploads/images/playlists/
ls -la backend/uploads/images/profiles/

# 6. Agregar los cambios del .gitignore:
git add .gitignore

# 7. Hacer commit de los cambios:
git commit -m "Remove uploads from git tracking, keep local files

- Updated .gitignore to ignore uploads content
- Removed tracked upload files from repository
- Local files preserved for development"

# 8. Push de los cambios:
git push origin develop

# 9. Verificar estado final:
git status
