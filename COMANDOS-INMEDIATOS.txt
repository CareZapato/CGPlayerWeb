# 🚨 COMANDOS INMEDIATOS - COPIAR Y PEGAR UNO POR UNO

# 1. Ver estado actual
docker ps

# 2. Verificar si hay archivos en uploads
docker exec cgplayer-app ls -la /app/backend/uploads/songs/

# 3. Buscar la canción específica
docker exec cgplayer-app find /app/backend/uploads/ -name "*wonderful_is_your_name*" -type f

# 4. Ver logs de errores recientes
docker logs cgplayer-app --tail=50 | grep -i error

# 5. Si no hay archivos, verificar el volumen
docker volume inspect cgplayerweb_app_uploads

# 6. Reiniciar la aplicación (mantiene la BD)
docker compose restart app

# 7. Verificar después del reinicio
docker exec cgplayer-app ls -la /app/backend/uploads/songs/

# 8. Si sigue sin archivos, recrear estructura básica
docker exec cgplayer-app mkdir -p /app/backend/uploads/songs /app/backend/uploads/events /app/backend/uploads/images/profiles /app/backend/uploads/images/playlists

# 9. Arreglar permisos
docker exec cgplayer-app chmod -R 755 /app/backend/uploads/
docker exec cgplayer-app chown -R cgplayer:nodejs /app/backend/uploads/

# 10. Ver logs en tiempo real para debuggear
docker compose logs -f app
