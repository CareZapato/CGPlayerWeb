# COMANDOS PARA ASEGURAR QUE TODO ESTÉ EN GIT

# 1. Verificar el estado actual:
git status

# 2. Agregar los archivos .gitkeep nuevos:
git add backend/uploads/songs/.gitkeep
git add backend/uploads/events/.gitkeep

# 3. Agregar el Dockerfile modificado:
git add Dockerfile

# 4. Agregar la documentación:
git add GARANTIAS-CARPETAS-UPLOADS.md

# 5. Hacer commit:
git commit -m "Ensure uploads folders always exist in Docker container

- Added .gitkeep files for songs/ and events/ folders
- Updated Dockerfile to create all upload subdirectories
- Added comprehensive folder structure documentation
- Guaranteed folder existence in VPS deployment"

# 6. Push:
git push origin develop

# 7. Verificar en el VPS después del pull:
# docker exec cgplayer-app ls -la /app/backend/uploads/
