# 🚨 COMANDOS RÁPIDOS PARA COPIAR Y PEGAR EN SSH

# ===== DIAGNÓSTICO RÁPIDO =====

# 1. Ver estado de contenedores
docker ps

# 2. Verificar uploads en el contenedor
docker exec cgplayer-app ls -la /app/backend/uploads/

# 3. Verificar carpeta songs específicamente  
docker exec cgplayer-app ls -la /app/backend/uploads/songs/

# 4. Buscar archivos de la canción problema
docker exec cgplayer-app find /app/backend/uploads/ -name "*wonderful_is_your_name*" -type f

# 5. Ver logs recientes del backup
docker logs cgplayer-app --tail=100 | grep -i backup

# 6. Verificar permisos de uploads
docker exec cgplayer-app stat /app/backend/uploads/songs/

# ===== SI NO HAY ARCHIVOS, VERIFICAR VOLÚMENES =====

# 7. Ver volúmenes de Docker
docker volume ls | grep cgplayer

# 8. Inspeccionar volumen de uploads
docker volume inspect cgplayerweb_app_uploads

# ===== COMANDOS DE EMERGENCIA SI ES NECESARIO =====

# 9. Reiniciar solo la aplicación (mantiene BD)
docker compose restart app

# 10. Ver logs en tiempo real
docker compose logs -f app
